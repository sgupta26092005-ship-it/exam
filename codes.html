Banker's Algorithm (Deadlock Avoidance) 
#include <stdio.h>

int main() {
    int alloc[5][3] = {{0,1,0},{2,0,0},{3,0,2},{2,1,1},{0,0,2}};
    int max[5][3] = {{7,5,3},{3,2,2},{9,0,2},{2,2,2},{4,3,3}};
    int avail[3] = {3,3,2};
    int need[5][3], finish[5]={0};
    int safe[5], count=0;

    for(int i=0;i<5;i++)
        for(int j=0;j<3;j++)
            need[i][j]=max[i][j]-alloc[i][j];

    while(count<5){
        for(int i=0;i<5;i++){
            if(!finish[i]){
                int flag=1;

                for(int j=0;j<3;j++)
                    if(need[i][j]>avail[j])
                        flag=0;

                if(flag){
                    for(int j=0;j<3;j++)
                        avail[j]+=alloc[i][j];

                    safe[count++]=i;
                    finish[i]=1;
                }
            }
        }
    }

    printf("Safe Sequence: ");

    for(int i=0;i<5;i++)
        printf("P%d ",safe[i]);

    return 0;
}
Producer Consumer using Semaphores 
#include <stdio.h>
#include <stdlib.h>

int mutex = 1, full = 0, empty = 3, item = 0;

int main() {
    int choice;

    while (1) {
        printf("\n1. Producer");
        printf("\n2. Consumer");
        printf("\n3. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &choice);

        switch (choice) {

        case 1:
            if (empty > 0 && mutex == 1) {
                mutex--;

                item++;
                full++;
                empty--;

                printf("Produced item %d\n", item);

                mutex++;
            }
            else {
                printf("Buffer Full!\n");
            }
            break;

        case 2:
            if (full > 0 && mutex == 1) {
                mutex--;

                printf("Consumed item %d\n", item);

                item--;
                full--;
                empty++;

                mutex++;
            }
            else {
                printf("Buffer Empty!\n");
            }
            break;

        case 3:
            exit(0);
        }
    }

    return 0;
}

    for(i=0; i<f; i++)

    for(i=0; i<n; i++)
    {
        found = 0;

        for(j=0; j<f; j++)

                found = 1;

        if(!found)
        {
            frames[pos] = pages[i];
            pos = (pos + 1) % f;
            faults++;
        }
    }

    printf("Page Faults = %d", faults);

    return 0;
}

Optimal Page Replacement 
#include <stdio.h>

int main()
{
    int frames[10], pages[30];
    int n, f, i, j, k, faults = 0;
    int found, pos, farthest;

    printf("Frames: ");
    scanf("%d", &f);

    printf("Number of pages: ");
    scanf("%d", &n);

    printf("Reference string: ");
    for(i=0; i<n; i++)
        scanf("%d", &pages[i]);

    for(i=0; i<f; i++)
        frames[i] = -1;

    for(i=0; i<n; i++)
    {
        found = 0;

        for(j=0; j<f; j++)
            if(frames[j] == pages[i])
                found = 1;

        if(!found)
        {
            pos = -1;
            farthest = i;

            for(j=0; j<f; j++)
            {
                for(k=i+1; k<n; k++)
                    if(frames[j] == pages[k])
                        break;

                if(k == n)
                {
                    pos = j;
                    break;
                }

                if(k > farthest)
                {
                    farthest = k;
                    pos = j;
                }
            }

            if(pos == -1)
                pos = 0;

            frames[pos] = pages[i];
            faults++;
        }
    }

    printf("Page Faults = %d", faults);

    return 0;
}

Message Queue - Sending Program 
#include <stdio.h>
#include <string.h>
#include <sys/msg.h>

struct msg {
    long type;
    char text[100];
};

int main()
{
    int msgid = msgget(12345, 0666 | IPC_CREAT);

    struct msg m;

    m.type = 1;

    while (1)
    {
        printf("Enter message: ");

        fgets(m.text, sizeof(m.text), stdin);

        m.text[strcspn(m.text, "\n")] = '\0';

        msgsnd(msgid, &m, sizeof(m.text), 0);

        if (strncmp(m.text, "end", 3) == 0)
            break;
    }

    return 0;
}

#include <stdio.h>

int main()
{
    int n, i;
    int bt[20], wt[20], tat[20];

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("P%d BT: ", i + 1);
        scanf("%d", &bt[i]);
    }

    wt[0] = 0;

    for(i = 1; i < n; i++)
        wt[i] = wt[i - 1] + bt[i - 1];

    for(i = 0; i < n; i++)
        tat[i] = wt[i] + bt[i];

    printf("\nProcess\tBT\tWT\tTAT\n");

    for(i = 0; i < n; i++)
        printf("P%d\t%d\t%d\t%d\n",
               i + 1, bt[i], wt[i], tat[i]);

    return 0;
}
FCFS Scheduling (With Arrival Time) 
#include <stdio.h>

int main()
{
    int n,i,j,temp;
    int p[20],at[20],bt[20],ct[20],tat[20],wt[20];

    scanf("%d",&n);

    for(i=0;i<n;i++)
    {
        p[i]=i+1;
        scanf("%d%d",&at[i],&bt[i]);
    }

    for(i=0;i<n-1;i++)
        for(j=i+1;j<n;j++)
            if(at[i]>at[j])
            {
                temp=at[i]; at[i]=at[j]; at[j]=temp;
                temp=bt[i]; bt[i]=bt[j]; bt[j]=temp;
                temp=p[i]; p[i]=p[j]; p[j]=temp;
            }

    ct[0]=at[0]+bt[0];

    for(i=1;i<n;i++)
    {
        if(ct[i-1]<at[i])
            ct[i]=at[i]+bt[i];
        else
            ct[i]=ct[i-1]+bt[i];
    }

    for(i=0;i<n;i++)
    {
        tat[i]=ct[i]-at[i];
        wt[i]=tat[i]-bt[i];
    }

    printf("\nPID\tAT\tBT\tCT\tTAT\tWT\n");

    for(i=0;i<n;i++)
        printf("P%d\t%d\t%d\t%d\t%d\t%d\n",
               p[i],at[i],bt[i],ct[i],tat[i],wt[i]);

    return 0;
}

Priority Scheduling (With Arrival Time) 
#include <stdio.h>

int main()
{
    int n,i,time=0,completed=0;
    int p[20],at[20],bt[20],pr[20];
    int ct[20],tat[20],wt[20];
    int visited[20]={0};

    scanf("%d",&n);

    for(i=0;i<n;i++)
    {
        p[i]=i+1;
        scanf("%d%d%d",&at[i],&bt[i],&pr[i]);
    }

    while(completed<n)
    {
        int index=-1;
        int high=9999;

        for(i=0;i<n;i++)
        {
            if(at[i]<=time &&
               visited[i]==0 &&
               pr[i]<high)
            {
                high=pr[i];
                index=i;
            }
        }

        if(index==-1)
            time++;
        else
        {
            time += bt[index];

            ct[index]=time;
            tat[index]=ct[index]-at[index];
            wt[index]=tat[index]-bt[index];

            visited[index]=1;
            completed++;
        }
    }

    printf("\nPID\tAT\tBT\tPR\tCT\tTAT\tWT\n");

    for(i=0;i<n;i++)
        printf("P%d\t%d\t%d\t%d\t%d\t%d\t%d\n",
               p[i],at[i],bt[i],pr[i],
               ct[i],tat[i],wt[i]);

    return 0;
}
Round Robin Scheduling 
#include <stdio.h>

int main()
{
    int n,tq,i,time=0,remain;
    int bt[20],rt[20],wt[20],tat[20];

    scanf("%d",&n);

    remain=n;

    for(i=0;i<n;i++)
    {
        scanf("%d",&bt[i]);
        rt[i]=bt[i];
    }

    scanf("%d",&tq);

    while(remain!=0)
    {
        for(i=0;i<n;i++)
        {
            if(rt[i]>0)
            {
                if(rt[i]<=tq)
                {
                    time += rt[i];

                    tat[i]=time;
                    wt[i]=tat[i]-bt[i];

                    rt[i]=0;
                    remain--;
                }
                else
                {
                    time += tq;
                    rt[i] -= tq;
                }
            }
        }
    }

    printf("\nPID\tBT\tWT\tTAT\n");

    for(i=0;i<n;i++)
        printf("P%d\t%d\t%d\t%d\n",
               i+1,bt[i],wt[i],tat[i]);

    return 0;
}
