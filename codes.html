#include <stdio.h>

int main() {
    int n, bt[20], wt[20], tat[20], i;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    printf("Enter Burst Times:\n");
    for(i=0;i<n;i++)
        scanf("%d",&bt[i]);

    wt[0]=0;
    for(i=1;i<n;i++)
        wt[i]=wt[i-1]+bt[i-1];

    for(i=0;i<n;i++)
        tat[i]=wt[i]+bt[i];

    printf("\nProcess BT WT TAT\n");
    for(i=0;i<n;i++)
        printf("%d\t%d\t%d\t%d\n",i+1,bt[i],wt[i],tat[i]);

    return 0;
}
1(b) opendir() readdir() 
#include <stdio.h>
#include <dirent.h>

int main() {
    DIR *dir;
    struct dirent *d;

    dir = opendir(".");

    if(dir == NULL) {
        printf("Cannot open directory");
        return 0;
    }

    while((d = readdir(dir)) != NULL)
        printf("%s\n", d->d_name);

    closedir(dir);
    return 0;
}

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>

int main() {
    int fd;
    char buf[20];

    fd = open("file.txt", O_RDONLY);

    if(fd < 0) {
        printf("File not found");
        return 0;
    }

    lseek(fd, 5, SEEK_SET);
    read(fd, buf, 10);

3(a) Round Robin 
#include <stdio.h>

int main() {
    int n, bt[20], rem[20], wt[20]={0}, tat[20];
    int tq, time=0, done, i;

    printf("Enter number of processes: ");
    scanf("%d",&n);

    printf("Enter Burst Times:\n");
    for(i=0;i<n;i++){
        scanf("%d",&bt[i]);
        rem[i]=bt[i];
    }

    printf("Enter Time Quantum: ");
    scanf("%d",&tq);

    do {
        done=1;
        for(i=0;i<n;i++){
            if(rem[i]>0){
                done=0;
                if(rem[i]>tq){
                    time+=tq;
                    rem[i]-=tq;
                }
                else{
                    time+=rem[i];
                    wt[i]=time-bt[i];
                    rem[i]=0;
                }
            }
        }
    } while(!done);

    for(i=0;i<n;i++)
        tat[i]=bt[i]+wt[i];

    printf("\nProcess BT WT TAT\n");
    for(i=0;i<n;i++)
        printf("%d\t%d\t%d\t%d\n",i+1,bt[i],wt[i],tat[i]);

    return 0;
}

6(a) Round Robin With Arrival Time 
#include <stdio.h>

int main() {
    int n, at[20], bt[20], rem[20], wt[20]={0}, tat[20];
    int tq, time=0, done, i;

    printf("Enter number of processes: ");
    scanf("%d",&n);

    printf("Enter Arrival Time and Burst Time:\n");
    for(i=0;i<n;i++){
        scanf("%d%d",&at[i],&bt[i]);
        rem[i]=bt[i];
    }

    printf("Enter Time Quantum: ");
    scanf("%d",&tq);

    do{
        done=1;
        for(i=0;i<n;i++){
            if(rem[i]>0 && at[i]<=time){
                done=0;
                if(rem[i]>tq){
                    time+=tq;
                    rem[i]-=tq;
                }
                else{
                    time+=rem[i];
                    wt[i]=time-bt[i]-at[i];
                    rem[i]=0;
                }
            }
        }
    }while(!done);

    for(i=0;i<n;i++)
        tat[i]=bt[i]+wt[i];

    printf("\nProcess AT BT WT TAT\n");
    for(i=0;i<n;i++)
        printf("%d\t%d\t%d\t%d\t%d\n",i+1,at[i],bt[i],wt[i],tat[i]);
}
