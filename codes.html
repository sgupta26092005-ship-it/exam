    float avg_wt = 0, avg_tat = 0;
    int done;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++) {
        printf("Enter burst time for process P%d: ", i+1);
        scanf("%d", &bt[i]);
        rem_bt[i] = bt[i];
        wt[i] = 0;
        tat[i] = 0;
    }

    printf("Enter Time Quantum: ");
    scanf("%d", &tq);

    do {
        done = 1;
        for(i = 0; i < n; i++) {
            if(rem_bt[i] > 0) {
                done = 0;
                if(rem_bt[i] > tq) {
                    rem_bt[i] -= tq;
                    time += tq;
                } else {
                    time += rem_bt[i];
                    wt[i] = time - bt[i];
                    rem_bt[i] = 0;
                }
            }
        }
    } while(!done);

    for(i = 0; i < n; i++) {
        tat[i] = wt[i] + bt[i];
        avg_wt += wt[i];
        avg_tat += tat[i];
    }

    avg_wt /= n;
    avg_tat /= n;

    printf("\nProcess\tBT\tWT\tTAT\n");
    for(i = 0; i < n; i++) {
        printf("P%d\t%d\t%d\t%d\n", i+1, bt[i], wt[i], tat[i]);
    }

    printf("\nAverage Waiting Time = %.2f", avg_wt);
    printf("\nAverage Turnaround Time = %.2f\n", avg_tat);

    return 0;
}

#include <stdio.h>

int main() {
    int n, tq, i, time = 0, done = 0;
    int at[10], bt[10], rem_bt[10], wt[10], tat[10];
    float avg_wt = 0, avg_tat = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++) {
        printf("\nProcess P%d\n", i+1);
        printf("Arrival Time: ");
        scanf("%d", &at[i]);
        printf("Burst Time: ");
        scanf("%d", &bt[i]);
        rem_bt[i] = bt[i];
        wt[i] = 0;
        tat[i] = 0;
    }

    printf("Enter Time Quantum: ");
    scanf("%d", &tq);

    int processes_remaining = n;

    while(processes_remaining > 0) {
        done = 1;

        for(i = 0; i < n; i++) {
            if(rem_bt[i] > 0 && at[i] <= time) {
                done = 0;

                if(rem_bt[i] > tq) {
                    rem_bt[i] -= tq;
                    time += tq;
                } else {
                    time += rem_bt[i];
                    wt[i] = time - at[i] - bt[i];
                    if(wt[i] < 0) wt[i] = 0;
                    rem_bt[i] = 0;
                    tat[i] = wt[i] + bt[i];
                    processes_remaining--;
                }
            }
        }

        if(done == 1)
            time++;
    }

    for(i = 0; i < n; i++) {
        avg_wt += wt[i];
        avg_tat += tat[i];
    }

    avg_wt /= n;
    avg_tat /= n;

    printf("\nProcess\tAT\tBT\tWT\tTAT\n");
    for(i = 0; i < n; i++)
        printf("P%d\t%d\t%d\t%d\t%d\n", i+1, at[i], bt[i], wt[i], tat[i]);

    printf("\nAverage Waiting Time = %.2f", avg_wt);
    printf("\nAverage Turnaround Time = %.2f\n", avg_tat);

    return 0;
}

{

    int total_wt = 0, total_tat = 0;
    float avg_wt, avg_tat;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("\nProcess %d\n", i+1);
        printf("Arrival Time: ");
        scanf("%d", &at[i]);
        printf("Burst Time: ");
        scanf("%d", &bt[i]);

        pid[i] = i + 1;
        completed[i] = 0;
    }

    int count = 0;

    while(count < n)
    {
        min = 9999;
        pos = -1;

        for(i = 0; i < n; i++)
        {
            if(at[i] <= time && completed[i] == 0)
            {
                if(bt[i] < min)
                {
                    min = bt[i];
                    pos = i;
                }
            }
        }

        if(pos == -1)
        {
            time++;
        }
        else
        {
            wt[pos] = time - at[pos];
            time += bt[pos];
            tat[pos] = wt[pos] + bt[pos];

            completed[pos] = 1;
            count++;

            total_wt += wt[pos];
            total_tat += tat[pos];
        }
    }

    avg_wt = (float) total_wt / n;
    avg_tat = (float) total_tat / n;

    printf("\nPID\tAT\tBT\tWT\tTAT\n");

FCFS With Arrival Time 
#include <stdio.h>

int main()
{
    int n, i, j, time = 0;
    int at[10], bt[10], wt[10], tat[10], ct[10], pid[10];
    float avg_wt = 0, avg_tat = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    // Input Arrival Time and Burst Time
    for(i = 0; i < n; i++)
    {
        printf("\nProcess P%d\n", i+1);
        printf("Arrival Time: ");
        scanf("%d", &at[i]);
        printf("Burst Time: ");
        scanf("%d", &bt[i]);
        pid[i] = i+1;
    }

    // Sort by Arrival Time
    for(i = 0; i < n-1; i++)
    {
        for(j = i+1; j < n; j++)
        {
            if(at[i] > at[j])
            {
                int temp;
                temp = at[i]; at[i] = at[j]; at[j] = temp;
                temp = bt[i]; bt[i] = bt[j]; bt[j] = temp;
                temp = pid[i]; pid[i] = pid[j]; pid[j] = temp;
            }
        }
    }

    // Completion Time
    ct[0] = at[0] + bt[0];
    for(i = 1; i < n; i++)
    {
        if(ct[i-1] < at[i])
            ct[i] = at[i] + bt[i];
        else
            ct[i] = ct[i-1] + bt[i];
    }

    // Waiting Time and Turnaround Time
    for(i = 0; i < n; i++)
    {
        tat[i] = ct[i] - at[i];
        wt[i] = tat[i] - bt[i];
        avg_wt += wt[i];
        avg_tat += tat[i];
    }

    avg_wt /= n;
    avg_tat /= n;

    // Display Output
    printf("\nPID\tAT\tBT\tWT\tTAT\n");
    for(i = 0; i < n; i++)
    {
        printf("P%d\t%d\t%d\t%d\t%d\n",
               pid[i], at[i], bt[i], wt[i], tat[i]);
    }

    printf("\nAverage Waiting Time = %.2f", avg_wt);
    printf("\nAverage Turnaround Time = %.2f\n", avg_tat);

    return 0;
}

#include <stdio.h>

int main() {
    int n, i, time = 0, completed = 0, highest;
    int at[10], bt[10], pr[10], wt[10], tat[10], pid[10], done[10];
    float avg_wt = 0, avg_tat = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++) {
        printf("\nProcess P%d\n", i+1);
        printf("Arrival Time: ");
        scanf("%d", &at[i]);
        printf("Burst Time: ");
        scanf("%d", &bt[i]);
        printf("Priority: ");
        scanf("%d", &pr[i]);

        pid[i] = i+1;
        done[i] = 0;
    }

    while(completed < n) {
        highest = -1;

        for(i = 0; i < n; i++) {
            if(at[i] <= time && done[i] == 0) {
                if(highest == -1 || pr[i] < pr[highest])
                    highest = i;
            }
        }

        if(highest == -1) {
            time++;
        } else {
            wt[highest] = time - at[highest];
            if(wt[highest] < 0) wt[highest] = 0;

            time += bt[highest];
            tat[highest] = wt[highest] + bt[highest];

            avg_wt += wt[highest];
            avg_tat += tat[highest];

            done[highest] = 1;
            completed++;
        }
    }

    printf("\nPID\tAT\tBT\tPR\tWT\tTAT\n");
    for(i = 0; i < n; i++)
        printf("P%d\t%d\t%d\t%d\t%d\t%d\n",
               pid[i], at[i], bt[i], pr[i], wt[i], tat[i]);

    printf("\nAverage Waiting Time = %.2f", avg_wt/n);
    printf("\nAverage Turnaround Time = %.2f\n", avg_tat/n);

    return 0;
}

#include <stdio.h>
#include <stdlib.h>
#include <dirent.h>

int main(int argc, char *argv[]) {
    DIR *dir;
    struct dirent *entry;

    if (argc < 2) {
        dir = opendir(".");
    } else {
        dir = opendir(argv[1]);
    }

    if (dir == NULL) {
        perror("opendir failed");
        return 1;
    }

    printf("Directory listing:\n");

    while ((entry = readdir(dir)) != NULL) {
        printf("%s\n", entry->d_name);
    }

    closedir(dir);

    return 0;
}

#include <fcntl.h>

#include <stdio.h>

int main() {
    int fd1, fd2;
    char buffer[100];
    int n;

    fd1 = open("input.txt", O_RDONLY);
    if (fd1 < 0) {
        printf("Error opening input file\n");
        return 1;
    }

    fd2 = open("output.txt", O_CREAT | O_WRONLY, 0644);
    if (fd2 < 0) {
        printf("Error creating output file\n");
        close(fd1);
        return 1;
    }

    while ((n = read(fd1, buffer, 100)) > 0) {
        write(fd2, buffer, n);
    }

    close(fd1);
    close(fd2);

    printf("File copied successfully.\n");

    return 0;
}
