Fixed Framing 
#include <stdio.h>

int main()
{
    char data[50], stuffed[50], destuffed[50];
    int frameSize, i, j = 0, position = 0;

    printf("Enter the data:\n");
    scanf("%s", data);

    printf("Enter the fsize:\n");
    scanf("%d", &frameSize);

    for (i = 0; data[position] != '\0'; i++)
    {
        if (i % (frameSize) == 0)
        {
            stuffed[j++] = '0' + frameSize;
        }
        else
        {
            stuffed[j++] = data[position++];
        }
    }

    stuffed[j] = '\0';

    printf("Stuffed Data : %s\n", stuffed);

    j = 0;
    for (i = 1; stuffed[i] != '\0'; i++)
        destuffed[j++] = stuffed[i];

    destuffed[j] = '\0';

    printf("DeStuffed Data : %s", destuffed);

    return 0;
}

Checksum 
#include <stdio.h>

int sender(int data[], int size)
{
    int sum = 0;
    printf("\n****SENDER SIDE****\n");

    for (int i = 0; i < size; i++)
        sum += data[i];

    printf("SUM IS: %d", sum);

    int checksum = ~sum;
    printf("\nCHECKSUM IS:%d", checksum);

    return checksum;
}

void receiver(int data[], int size, int senderChecksum)
{
    int sum = 0;
    printf("\n\n****RECEIVER SIDE****\n");

    for (int i = 0; i < size; i++)
        sum += data[i];

    printf("SUM IS:%d", sum);

    int checksum = ~(sum + senderChecksum);
    printf("\nCHECKSUM IS:%d", checksum);
}

int main()
{
    int size;
    printf("\nENTER SIZE OF THE STRING:");
    scanf("%d", &size);

    int data[size];
    printf("ENTER THE ELEMENTS OF THE ARRAY TO CALCULATE CHECKSUM:\n");

    for (int i = 0; i < size; i++)
        scanf("%d", &data[i]);

    int senderChecksum = sender(data, size);
    receiver(data, size, senderChecksum);

    return 0;
}
CRC - Cyclic Redundancy Check 
#include <stdio.h>
#include <string.h>

int main()
{
    char message[50], generator[20], temp[20];
    int messageLength, generatorLength, error = 0;

    printf("\n enter message");
    gets(message);

    messageLength = strlen(message);
    printf("\n msg length:%d", messageLength);

    printf("\n enter generator or divisor");
    gets(generator);

    generatorLength = strlen(generator);
    printf("\n generator msg length:%d", generatorLength);

    // Add zeros
    for (int i = 0; i < generatorLength - 1; i++)
        message[messageLength + i] = '0';

    message[messageLength + generatorLength - 1] = '\0';

    printf("\n after padding,msg is:%s", message);

    // CRC division
    for (int i = 0; i < generatorLength; i++)
        temp[i] = message[i];

    for (int i = 0; i < messageLength; i++)
    {
        if (temp[0] == '1')
        {
            for (int j = 0; j < generatorLength; j++)
                temp[j] = (temp[j] == generator[j]) ? '0' : '1';
        }

        for (int j = 0; j < generatorLength - 1; j++)
            temp[j] = temp[j + 1];

        temp[generatorLength - 1] = message[i + generatorLength];
    }

    temp[generatorLength - 1] = '\0';

    printf("\n remainder is:%s", temp);

    for (int i = 0; i < generatorLength - 1; i++)
        message[messageLength + i] = temp[i];

    message[messageLength + generatorLength - 1] = '\0';

    printf("\n after adding remainder,msg is:%s", message);

    printf("\n enter the received message");
    gets(message);

    // Check received message
    for (int i = 0; i < generatorLength; i++)
        temp[i] = message[i];

    for (int i = 0; i < messageLength; i++)
    {
        if (temp[0] == '1')
        {
            for (int j = 0; j < generatorLength; j++)
                temp[j] = (temp[j] == generator[j]) ? '0' : '1';
        }

        for (int j = 0; j < generatorLength - 1; j++)
            temp[j] = temp[j + 1];

        temp[generatorLength - 1] = message[i + generatorLength];
    }

    temp[generatorLength - 1] = '\0';

    printf("\n remainder is:%s", temp);

    for (int i = 0; i < generatorLength - 1; i++)
    {
        if (temp[i] != '0')
        {
            error = 1;
            break;
        }
    }

    if (error == 0)
        printf("\n no error in received message");
    else
        printf("\n error in received message");

    return 0;
}
Go-Back-N Algorithm 
#include <stdio.h>

int main()
{
    int windowSize, sent = 0, acknowledgement;

    printf("enter window size\n");
    scanf("%d", &windowSize);

    while (1)
    {
        for (int i = 0; i < windowSize; i++)
        {
            printf("Frame %d has been transmitted.\n", sent);
            sent++;

            if (sent == windowSize)
                break;
        }

        printf("\nPlease enter the last Acknowledgement received.\n");
        scanf("%d", &acknowledgement);

        if (acknowledgement == windowSize)
            break;

        sent = acknowledgement;
    }

    return 0;
}
IP Address Class 
#include <stdio.h>

int main()
{
    int a, b, c, d;

    printf("Enter an IP Address: ");
    scanf("%d.%d.%d.%d", &a, &b, &c, &d);

    printf("IP Address: %d.%d.%d.%d\n", a, b, c, d);

    if (a < 1 || a > 255 || b < 0 || b > 255 ||
        c < 0 || c > 255 || d < 0 || d > 255)
    {
        printf("Reserved IP Address\n");
    }
    else if (a >= 1 && a <= 126)
    {
        printf("Class: A\n");
        printf("Network ID: %d.0.0.0\n", a);
        printf("Host ID: 0.%d.%d.%d\n", b, c, d);
        printf("Default Subnet Mask: 255.0.0.0\n");
    }
    else if (a >= 128 && a <= 191)
    {
        printf("Class: B\n");
        printf("Network ID: %d.%d.0.0\n", a, b);
        printf("Host ID: 0.0.%d.%d\n", c, d);
        printf("Default Subnet Mask: 255.255.0.0\n");
    }
    else if (a >= 192 && a <= 223)
    {
        printf("Class: C\n");
        printf("Network ID: %d.%d.%d.0\n", a, b, c);
        printf("Host ID: 0.0.0.%d\n", d);
        printf("Default Subnet Mask: 255.255.255.0\n");
    }
    else if (a >= 224 && a <= 239)
    {
        printf("Class: D\n");
        printf("Class D is used for multicasting.\n");
    }
    else if (a >= 240 && a <= 255)
    {
        printf("Class: E\n");
        printf("Class E is reserved for experimental use.\n");
    }
    else
    {
        printf("Reserved IP Address\n");
    }

    return 0;
}
