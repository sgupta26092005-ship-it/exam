Banker's Algorithm (Deadlock Avoidance) 
#include <stdio.h>

int main() {
    int alloc[5][3] = {{0,1,0},{2,0,0},{3,0,2},{2,1,1},{0,0,2}};
    int max[5][3] = {{7,5,3},{3,2,2},{9,0,2},{2,2,2},{4,3,3}};
    int avail[3] = {3,3,2};
    int need[5][3], finish[5]={0};
    int safe[5], count=0;

    for(int i=0;i<5;i++)
        for(int j=0;j<3;j++)
            need[i][j]=max[i][j]-alloc[i][j];

    while(count<5){
        for(int i=0;i<5;i++){
            if(!finish[i]){
                int flag=1;

                for(int j=0;j<3;j++)
                    if(need[i][j]>avail[j])
                        flag=0;

                if(flag){
                    for(int j=0;j<3;j++)
                        avail[j]+=alloc[i][j];

                    safe[count++]=i;
                    finish[i]=1;
                }
            }
        }
    }

    printf("Safe Sequence: ");

    for(int i=0;i<5;i++)
        printf("P%d ",safe[i]);

    return 0;
}
Producer Consumer using Semaphores 
#include <stdio.h>
#include <stdlib.h>

int mutex = 1, full = 0, empty = 3, item = 0;

int main() {
    int choice;

    while (1) {
        printf("\n1. Producer");
        printf("\n2. Consumer");
        printf("\n3. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &choice);

        switch (choice) {

        case 1:
            if (empty > 0 && mutex == 1) {
                mutex--;

                item++;
                full++;
                empty--;

                printf("Produced item %d\n", item);

                mutex++;
            }
            else {
                printf("Buffer Full!\n");
            }
            break;

        case 2:
            if (full > 0 && mutex == 1) {
                mutex--;

                printf("Consumed item %d\n", item);

                item--;
                full--;
                empty++;

                mutex++;
            }
            else {
                printf("Buffer Empty!\n");
            }
            break;

        case 3:
            exit(0);
        }
    }

    return 0;
}

#include <stdio.h>

int main()
{
    int frames[10], pages[30];
    int n, f, i, j, pos = 0, faults = 0, found;

    printf("Frames: ");
    scanf("%d", &f);

    printf("Number of pages: ");
    scanf("%d", &n);

    printf("Reference string: ");
    for(i=0; i<n; i++)
        scanf("%d", &pages[i]);

    for(i=0; i<f; i++)
        frames[i] = -1;

    for(i=0; i<n; i++)
    {
        found = 0;

        for(j=0; j<f; j++)
            if(frames[j] == pages[i])
                found = 1;

        if(!found)
        {
            frames[pos] = pages[i];
            pos = (pos + 1) % f;
            faults++;
        }
    }

    printf("Page Faults = %d", faults);

    return 0;
}

#include <stdio.h>

int main()
{
    int frames[10], pages[30], recent[10];
    int n, f, i, j, faults = 0, found, pos;

    printf("Frames: ");
    scanf("%d", &f);

    printf("Number of pages: ");
    scanf("%d", &n);

    printf("Reference string: ");
    for(i=0; i<n; i++)
        scanf("%d", &pages[i]);

    for(i=0; i<f; i++)
        frames[i] = -1;

    for(i=0; i<n; i++)
    {
        found = 0;

        for(j=0; j<f; j++)
        {
            if(frames[j] == pages[i])
            {
                found = 1;
                recent[j] = i;
            }
        }

        if(!found)
        {
            pos = 0;

            for(j=1; j<f; j++)
            {
                if(frames[j] == -1)
                {
                    pos = j;
                    break;
                }

                if(recent[j] < recent[pos])
                    pos = j;
            }

            frames[pos] = pages[i];
            recent[pos] = i;
            faults++;
        }
    }

    printf("Page Faults = %d", faults);

    return 0;
}
Optimal Page Replacement 
#include <stdio.h>

int main()
{
    int frames[10], pages[30];
    int n, f, i, j, k, faults = 0;
    int found, pos, farthest;

    printf("Frames: ");
    scanf("%d", &f);

    printf("Number of pages: ");
    scanf("%d", &n);

    printf("Reference string: ");
    for(i=0; i<n; i++)
        scanf("%d", &pages[i]);

    for(i=0; i<f; i++)
        frames[i] = -1;

    for(i=0; i<n; i++)
    {
        found = 0;

        for(j=0; j<f; j++)
            if(frames[j] == pages[i])
                found = 1;

        if(!found)
        {
            pos = -1;
            farthest = i;

            for(j=0; j<f; j++)
            {
                for(k=i+1; k<n; k++)
                    if(frames[j] == pages[k])
                        break;

                if(k == n)
                {
                    pos = j;
                    break;
                }

                if(k > farthest)
                {
                    farthest = k;
                    pos = j;
                }
            }

            if(pos == -1)
                pos = 0;

            frames[pos] = pages[i];
            faults++;
        }
    }

    printf("Page Faults = %d", faults);

    return 0;
}

{

Message Queue - Sending Program 
#include <stdio.h>
#include <string.h>
#include <sys/msg.h>

struct msg {
    long type;
    char text[100];
};

int main()
{
    int msgid = msgget(12345, 0666 | IPC_CREAT);

    struct msg m;

    m.type = 1;

    while (1)
    {
        printf("Enter message: ");

        fgets(m.text, sizeof(m.text), stdin);

        m.text[strcspn(m.text, "\n")] = '\0';

        msgsnd(msgid, &m, sizeof(m.text), 0);

        if (strncmp(m.text, "end", 3) == 0)
            break;
    }

    return 0;
}
Message Queue - Receiving Program 
#include <stdio.h>
#include <string.h>
#include <sys/msg.h>

struct msg {
    long type;
    char text[100];
};

int main()
{
    int msgid = msgget(12345, 0666 | IPC_CREAT);

    struct msg m;

    while (1)
    {
        msgrcv(msgid, &m, 100, 0, 0);

        printf("Received: %s\n", m.text);

        if (strncmp(m.text, "end", 3) == 0)
            break;
    }

    msgctl(msgid, IPC_RMID, NULL);

    return 0;
}
Segmentation Technique 
#include <stdio.h>

int main()
{
    int n, limit[10], base[10];
    int seg, off, i;

    printf("Number of segments: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("Limit and Base of segment %d: ", i);
        scanf("%d%d", &limit[i], &base[i]);
    }

    while(1)
    {
        printf("Enter segment number and offset: ");
        scanf("%d%d", &seg, &off);

        if(seg >= n)
            printf("Invalid Segment\n");
        else if(off >= limit[seg])
            printf("Invalid Offset\n");
        else
            printf("Physical Address = %d\n", base[seg] + off);
    }

    return 0;
}

FIFO Program 1 - Writes First Then Reads 
#include <stdio.h>
#include <string.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <sys/types.h>
#include <unistd.h>

int main()
{
    int fd;

    char *myfifo = "/home/student/kk_22/myfifo";

    mkfifo(myfifo, 0666);

    char arr1[80], arr2[80];

    while (1)
    {
        fgets(arr2, 80, stdin);

        fd = open(myfifo, O_WRONLY);

        write(fd, arr2, strlen(arr2)+1);

        close(fd);

        fd = open(myfifo, O_RDONLY);

        read(fd, arr1, sizeof(arr1));

        printf("User2: %s\n", arr1);

        close(fd);
    }

    return 0;
}
