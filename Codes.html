6(a) Round Robin without Arrival Time

#include 

int main() {
    int n, i, time = 0, tq;
    int bt[10], rem_bt[10], wt[10], tat[10];
    float avg_wt = 0, avg_tat = 0;
    int done;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++) {
        printf("Enter burst time for process P%d: ", i+1);
        scanf("%d", &bt[i]);
        rem_bt[i] = bt[i];
    }

    printf("Enter Time Quantum: ");
    scanf("%d", &tq);

    do {
        done = 1;
        for(i = 0; i < n; i++) {
            if(rem_bt[i] > 0) {
                done = 0;
                if(rem_bt[i] > tq) {
                    rem_bt[i] -= tq;
                    time += tq;
                } else {
                    time += rem_bt[i];
                    wt[i] = time - bt[i];
                    rem_bt[i] = 0;
                }
            }
        }
    } while(!done);

    for(i = 0; i < n; i++) {
        tat[i] = wt[i] + bt[i];
        avg_wt += wt[i];
        avg_tat += tat[i];
    }

    avg_wt /= n;
    avg_tat /= n;

    printf("\nProcess\tBT\tWT\tTAT\n");
    for(i = 0; i < n; i++)
        printf("P%d\t%d\t%d\t%d\n", i+1, bt[i], wt[i], tat[i]);

    printf("\nAverage WT = %.2f", avg_wt);
    printf("\nAverage TAT = %.2f\n", avg_tat);

    return 0;
}

6(b) Round Robin with Arrival Time

#include 

int main() {
    int n, tq, i, time = 0, done;
    int at[10], bt[10], rem_bt[10], wt[10], tat[10];
    float avg_wt = 0, avg_tat = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++) {
        printf("\nProcess P%d\n", i+1);
        printf("Arrival Time: ");
        scanf("%d", &at[i]);
        printf("Burst Time: ");
        scanf("%d", &bt[i]);
        rem_bt[i] = bt[i];
    }

    printf("Enter Time Quantum: ");
    scanf("%d", &tq);

    int remaining = n;

    while(remaining > 0) {
        done = 1;

        for(i = 0; i < n; i++) {
            if(rem_bt[i] > 0 && at[i] <= time) {
                done = 0;

                if(rem_bt[i] > tq) {
                    rem_bt[i] -= tq;
                    time += tq;
                } else {
                    time += rem_bt[i];
                    wt[i] = time - at[i] - bt[i];
                    if(wt[i] < 0) wt[i] = 0;
                    tat[i] = wt[i] + bt[i];
                    rem_bt[i] = 0;
                    remaining--;
                }
            }
        }

        if(done)
            time++;
    }

    for(i = 0; i < n; i++) {
        avg_wt += wt[i];
        avg_tat += tat[i];
    }

    printf("\nProcess\tAT\tBT\tWT\tTAT\n");
    for(i = 0; i < n; i++)
        printf("P%d\t%d\t%d\t%d\t%d\n", i+1, at[i], bt[i], wt[i], tat[i]);

    printf("\nAverage WT = %.2f", avg_wt/n);
    printf("\nAverage TAT = %.2f\n", avg_tat/n);

    return 0;
}

6(c) SJF with Arrival Time

#include 

int main() {
    int n, i, time = 0, min, pos;
    int at[10], bt[10], wt[10], tat[10], completed[10] = {0};
    int count = 0;
    float avg_wt = 0, avg_tat = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++) {
        printf("\nProcess %d\n", i+1);
        printf("Arrival Time: ");
        scanf("%d", &at[i]);
        printf("Burst Time: ");
        scanf("%d", &bt[i]);
    }

    while(count < n) {
        min = 9999;
        pos = -1;

        for(i = 0; i < n; i++) {
            if(at[i] <= time && completed[i] == 0 && bt[i] < min) {
                min = bt[i];
                pos = i;
            }
        }

        if(pos == -1)
            time++;
        else {
            wt[pos] = time - at[pos];
            if(wt[pos] < 0) wt[pos] = 0;

            time += bt[pos];
            tat[pos] = wt[pos] + bt[pos];

            avg_wt += wt[pos];
            avg_tat += tat[pos];

            completed[pos] = 1;
            count++;
        }
    }

    printf("\nProcess\tAT\tBT\tWT\tTAT\n");
    for(i = 0; i < n; i++)
        printf("P%d\t%d\t%d\t%d\t%d\n", i+1, at[i], bt[i], wt[i], tat[i]);

    printf("\nAverage WT = %.2f", avg_wt/n);
    printf("\nAverage TAT = %.2f\n", avg_tat/n);

    return 0;
}

6(d) SJF without Arrival Time

#include 

int main() {
    int n, i, j, temp;
    int bt[10], wt[10], tat[10];

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
        scanf("%d", &bt[i]);

    for(i = 0; i < n; i++)
        for(j = i+1; j < n; j++)
            if(bt[i] > bt[j]) {
                temp = bt[i];
                bt[i] = bt[j];
                bt[j] = temp;
            }

    wt[0] = 0;
    for(i = 1; i < n; i++)
        wt[i] = wt[i-1] + bt[i-1];

    for(i = 0; i < n; i++)
        tat[i] = wt[i] + bt[i];

    printf("\nProcess\tBT\tWT\tTAT\n");
    for(i = 0; i < n; i++)
        printf("%d\t%d\t%d\t%d\n", i+1, bt[i], wt[i], tat[i]);

    return 0;
}

6(g) Priority with Arrival Time

#include 

int main() {
    int n, i, time = 0, completed = 0, highest;
    int at[10], bt[10], pr[10], wt[10], tat[10], done[10] = {0};
    float avg_wt = 0, avg_tat = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
        scanf("%d%d%d", &at[i], &bt[i], &pr[i]);

    while(completed < n) {
        highest = -1;

        for(i = 0; i < n; i++) {
            if(at[i] <= time && done[i] == 0) {
                if(highest == -1 || pr[i] < pr[highest])
                    highest = i;
            }
        }

        if(highest == -1)
            time++;
        else {
            wt[highest] = time - at[highest];
            if(wt[highest] < 0) wt[highest] = 0;

            time += bt[highest];
            tat[highest] = wt[highest] + bt[highest];

            avg_wt += wt[highest];
            avg_tat += tat[highest];

            done[highest] = 1;
            completed++;
        }
    }

    printf("\nProcess\tAT\tBT\tPR\tWT\tTAT\n");
    for(i = 0; i < n; i++)
        printf("%d\t%d\t%d\t%d\t%d\t%d\n", i+1, at[i], bt[i], pr[i], wt[i], tat[i]);

    printf("\nAverage WT = %.2f", avg_wt/n);
    printf("\nAverage TAT = %.2f\n", avg_tat/n);

    return 0;
}
